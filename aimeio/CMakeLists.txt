cmake_minimum_required(VERSION 3.21)
project(SKPassAimeIO C CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(skpass_hid STATIC ../shared/windows_hid.c ../esp32/main/hid_wire.c)
target_include_directories(skpass_hid PUBLIC ../shared ../esp32/main)
target_link_libraries(skpass_hid PUBLIC setupapi hid)
add_library(aimeio SHARED aimeio.c cards_json.c spad.cpp)
if(MINGW)
    add_link_options(-static-libgcc -static-libstdc++)
    target_link_options(aimeio PRIVATE -static-libgcc -static-libstdc++)
endif()
target_link_libraries(aimeio PRIVATE skpass_hid setupapi user32)
set_target_properties(aimeio PROPERTIES PREFIX "" OUTPUT_NAME "aimeio_skpass")
include(CTest)
if(BUILD_TESTING)
    add_executable(aimeio_tests tests.c cards_json.c spad.cpp)
    target_link_libraries(aimeio_tests PRIVATE skpass_hid setupapi user32)
    add_test(NAME aimeio_frames COMMAND aimeio_tests)
    add_executable(aimeio_dll_check dll_check.c)
    add_test(NAME aimeio_actual_dll COMMAND aimeio_dll_check $<TARGET_FILE:aimeio>)
endif()
