cmake_minimum_required(VERSION 3.21)
project(SKPass VERSION 1.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core Widgets SerialPort)
add_library(skpass_hid STATIC ../shared/windows_hid.c ../esp32/main/hid_wire.c)
target_include_directories(skpass_hid PUBLIC ../shared ../esp32/main)
target_link_libraries(skpass_hid PUBLIC setupapi hid)
add_library(skpass_core src/AimeCode.cpp src/Spad.cpp src/FrameParser.cpp src/WriteProtocol.cpp src/FirmwareImage.cpp src/RomDownload.cpp src/NfcDetails.cpp src/core/ReaderProtocol.cpp ../esp32/main/advanced_protocol.c ../esp32/main/card_emulation.c ../esp32/main/ce_protocol.c)
target_include_directories(skpass_core PUBLIC src ../esp32/main)
target_link_libraries(skpass_core PUBLIC Qt6::Core Qt6::SerialPort skpass_hid)
add_library(skpass_update STATIC src/FirmwareUpdatePage.cpp src/RomDownloadPanel.cpp src/NfcPage.cpp src/AdvancedPanel.cpp src/CardEmulationPanel.cpp)
target_link_libraries(skpass_update PUBLIC skpass_core Qt6::Widgets)
add_executable(skpass_desktop WIN32 src/main.cpp src/MainWindow.cpp src/SerialReader.cpp src/CardManager.cpp src/SegatoolsEditor.cpp src/aimeio_resources.qrc src/app.rc)
set_target_properties(skpass_desktop PROPERTIES OUTPUT_NAME "SKPass")
add_custom_command(TARGET skpass_desktop POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/tools/esptool/esptool" "$<TARGET_FILE_DIR:skpass_desktop>/tools/esptool"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../releases/v1.0/firmware/first-flash" "$<TARGET_FILE_DIR:skpass_desktop>/tools/first-flash")
target_link_libraries(skpass_desktop PRIVATE skpass_update Qt6::Widgets Qt6::SerialPort)
if(MSVC)
    target_compile_options(skpass_core PRIVATE /utf-8)
    target_compile_options(skpass_desktop PRIVATE /utf-8)
endif()
