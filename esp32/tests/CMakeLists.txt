cmake_minimum_required(VERSION 3.21)
project(SKPassFirmwareMocks LANGUAGES C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_executable(card_emulation_tests card_emulation_tests.c ../main/card_emulation.c ../main/ce_protocol.c)
target_include_directories(card_emulation_tests PRIVATE ../main)
add_test(NAME t4t_ndef_and_ce_protocol COMMAND card_emulation_tests)
add_executable(rom_download_tests rom_download_tests.c)
target_include_directories(rom_download_tests PRIVATE rom_stubs ../main)
add_test(NAME rom_download_entry COMMAND rom_download_tests)
add_executable(uf2_update_tests uf2_update_tests.c)
target_include_directories(uf2_update_tests PRIVATE uf2_stubs ../main)
add_test(NAME uf2_update_restart COMMAND uf2_update_tests)
add_executable(hid_wire_tests hid_wire_tests.c ../main/hid_wire.c ../main/reader_frame.c)
target_include_directories(hid_wire_tests PRIVATE ../main)
add_test(NAME vendor_hid_framing COMMAND hid_wire_tests)
add_executable(portable_tests portable_tests.c ../main/reader_frame.c ../components/skpass/SKPass.c)
target_include_directories(portable_tests PRIVATE ../main ../components/skpass)
add_test(NAME hid_and_crc COMMAND portable_tests)
add_executable(mifare_reader_tests mifare_reader_tests.c ../main/mifare_reader.c ../main/aime_reader.c)
target_include_directories(mifare_reader_tests PRIVATE mifare_stubs ../main)
add_test(NAME mifare_retries COMMAND mifare_reader_tests)
add_executable(writer_tests writer_tests.c ../main/card_writer.c ../main/writer_protocol.c ../main/reader_frame.c)
target_include_directories(writer_tests PRIVATE . ../main)
if(MSVC)
    target_compile_options(writer_tests PRIVATE /FIwriter_platform.h)
else()
    target_compile_options(writer_tests PRIVATE -include writer_platform.h)
endif()
add_test(NAME bounded_writer COMMAND writer_tests)
add_executable(smartcard_probe_tests smartcard_probe_tests.c)
target_include_directories(smartcard_probe_tests PRIVATE pn7150_stubs ../main)
add_test(NAME smartcard_public_probe_vectors COMMAND smartcard_probe_tests)
add_executable(advanced_tests advanced_tests.c ../main/advanced_reader.c ../main/advanced_protocol.c)
target_include_directories(advanced_tests PRIVATE pn7150_stubs ../main)
if(NOT MSVC)
    target_compile_options(advanced_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME advanced_typed_readwrite COMMAND advanced_tests)
add_executable(advanced_session_tests advanced_session_tests.c ../main/advanced_protocol.c ../main/reader_frame.c ../main/hid_wire.c ../main/writer_protocol.c ../main/card_emulation.c ../main/ce_protocol.c)
target_include_directories(advanced_session_tests PRIVATE cdc_stubs pn7150_stubs ../main)
add_test(NAME advanced_cdc_hid_session COMMAND advanced_session_tests)
add_executable(pn7150_tests pn7150_tests.c)
target_include_directories(pn7150_tests PRIVATE pn7150_stubs ../main)
if(NOT MSVC)
    target_compile_options(pn7150_tests PRIVATE -Wall -Wextra -Werror)
endif()
foreach(suite irq i2c discovery initialization emulation)
    add_test(NAME pn7150_${suite} COMMAND pn7150_tests ${suite})
endforeach()
